    function escHtml(str) {
      return String(str ?? "")
        .replaceAll("&", "&amp;")
        .replaceAll("<", "&lt;")
        .replaceAll(">", "&gt;")
        .replaceAll('"', "&quot;")
        .replaceAll("'", "&#039;");
    }

    function slugify(text) {
      return String(text || "")
        .toLowerCase()
        .trim()
        .replace(/[^a-z0-9]+/g, "-")
        .replace(/^-+|-+$/g, "") || "section";

    function getCssFontStack(preset, uploadedFallbackName) {
      if (uploadedFallbackName) return `"${uploadedFallbackName}", ${presetToStack(preset)}`;
      return presetToStack(preset);
    }

    function presetToStack(preset) {
      if (preset === "garamond") return `"EB Garamond","Garamond","Cormorant Garamond","Times New Roman",serif`;
      if (preset === "baskerville") return `"Libre Baskerville","Baskerville","Baskerville Old Face","Times New Roman",serif`;
      return `"Georgia","Times New Roman",serif`;
    }

      const displayStack = getCssFontStack(els.displayFontPreset.value, state.currentFontFaces.display);

      document.documentElement.style.setProperty("--body-font", bodyStack);
      document.documentElement.style.setProperty("--display-font", displayStack);
    }

    async function embedFont(file, key, fontFamilyName) {
      if (!file) return null;
      const base64 = await fileToDataUrl(file);
      const format =
        file.name.endsWith(".woff2") ? "woff2" :
        file.name.endsWith(".woff") ? "woff" :
        file.name.endsWith(".otf") ? "opentype" :
        "truetype";

      const styleId = `embedded-font-${key}`;
      const existing = document.getElementById(styleId);
      if (existing) existing.remove();

      const style = document.createElement("style");
      style.id = styleId;
      style.textContent = `
        @font-face {
          font-family: "${fontFamilyName}";
          src: url("${base64}") format("${format}");
          font-weight: normal;
          font-style: normal;
          font-display: swap;
        }
      `;
      document.head.appendChild(style);
      state.currentFontFaces[key] = fontFamilyName;
      return fontFamilyName;
    }

    function fileToDataUrl(file) {
      return new Promise((resolve, reject) => {
        const reader = new FileReader();
        reader.onload = () => resolve(String(reader.result));
        reader.onerror = reject;
        reader.readAsDataURL(file);
      });
    }

    function stripHtml(html = "") {
      const div = document.createElement("div");
      div.innerHTML = html;
      return div.textContent || div.innerText || "";
    }

    function countWords(html = "") {
      const text = stripHtml(html).replace(/\s+/g, " ").trim();
      if (!text) return 0;
      return text.split(" ").length;
    }

    function parseManuscript(raw) {
      const cleaned = String(raw || "").replace(/\r\n/g, "\n").trim();
      if (!cleaned) return [];

      // HTML-aware split: if content includes <p> tags, turn them into blocks first.
      if (/<p[\s>]/i.test(cleaned) || /<div[\s>]/i.test(cleaned)) {
        return parseHtmlManuscript(cleaned);
      }

      return parsePlainTextManuscript(cleaned);
    }

    function parsePlainTextManuscript(text) {
      const chunks = text.split(/\n{2,}/).map(s => s.trim()).filter(Boolean);
      const blocks = [];

      for (const chunk of chunks) {
        if (/^\*{3,}$/.test(chunk) || /^# # #$/.test(chunk)) {
          blocks.push({ type: "scene" });
          continue;
        }

        if (/^(chapter|prologue|epilogue)\b/i.test(chunk)) {
          blocks.push({ type: "chapter", text: chunk });
          continue;
        }

        blocks.push({ type: "paragraph", html: escHtml(chunk).replace(/\n/g, "<br>") });
      }

      return blocks;
    }

    function parseHtmlManuscript(html) {
      const wrapper = document.createElement("div");
      wrapper.innerHTML = html;

      const blocks = [];
      const nodes = Array.from(wrapper.childNodes);

      nodes.forEach(node => {
        if (node.nodeType === Node.TEXT_NODE) {
          const raw = node.textContent.trim();
          if (!raw) return;
          if (/^\*{3,}$/.test(raw)) {
            blocks.push({ type: "scene" });
          } else if (/^(chapter|prologue|epilogue)\b/i.test(raw)) {
            blocks.push({ type: "chapter", text: raw });
          } else {
            blocks.push({ type: "paragraph", html: escHtml(raw) });
          }
          return;
        }

        if (node.nodeType !== Node.ELEMENT_NODE) return;
        const tag = node.tagName.toLowerCase();
        const text = (node.textContent || "").trim();

        if (!text && tag !== "img") return;

        if (/^(chapter|prologue|epilogue)\b/i.test(text)) {
          blocks.push({ type: "chapter", text });
          return;
        }

        if (/^\*{3,}$/.test(text)) {
          blocks.push({ type: "scene" });
          return;
        }

        if (tag === "p" || tag === "div" || tag === "blockquote") {
          blocks.push({ type: "paragraph", html: node.outerHTML });
        } else if (/^h[1-4]$/.test(tag)) {
          blocks.push({ type: "chapter", text });
        } else {
          blocks.push({ type: "paragraph", html: node.outerHTML });
        }
      });

      return blocks;
    }

    function buildSections() {
      const project = getProjectData();
      const blocks = parseManuscript(project.manuscript);

      const sections = [];
      const tocEntries = [];

      if (project.layout.includeTitlePage) {
        sections.push({
          id: "title-page",
          kind: "front",
          headerLeft: "",
          headerRight: "",
          showHeader: false,
          showFooter: false,
          html: renderTitlePage(project.metadata)
        });
      }

      if (project.layout.includeCopyrightPage) {
        sections.push({
          id: "copyright-page",
          kind: "front",
          headerLeft: "",
          headerRight: "",
          showHeader: false,
          showFooter: false,
          html: renderCopyrightPage(project.metadata)
        });
      }

      if (project.layout.includeDedicationPage && project.metadata.dedication.trim()) {
        sections.push({
          id: "dedication-page",
          kind: "front",
          headerLeft: "",
          headerRight: "",
          showHeader: false,
          showFooter: false,
          html: renderDedicationPage(project.metadata.dedication)
        });
      }

      const chapterGroups = [];
      let current = null;

      blocks.forEach((block, index) => {
        if (block.type === "chapter") {
          if (current) chapterGroups.push(current);
          current = {
            title: block.text,
            id: `chapter-${chapterGroups.length + 1}-${slugify(block.text)}`,
            blocks: []
          };
          return;
        }

        if (!current) {
          current = {
            title: "Opening",
            id: "opening",
            blocks: []
          };
        }

        current.blocks.push(block);
      });

      if (current) chapterGroups.push(current);

      chapterGroups.forEach((chapter, idx) => {
        tocEntries.push({
          id: chapter.id,
          title: chapter.title,
          order: idx + 1
        });

        sections.push({
          id: chapter.id,
          kind: "chapter",
          chapterTitle: chapter.title,
          headerLeft: project.metadata.author || "",
          headerRight: project.metadata.title || "",
          showHeader: true,
          showFooter: true,
          html: renderChapterSection(chapter, idx === 0)
        });
      });

      if (project.layout.includeTOC && tocEntries.length) {
        const tocIndex = sections.findIndex(s => s.kind === "chapter");
        const tocSection = {
          id: "table-of-contents",
          kind: "front",
          headerLeft: "",
          headerRight: "",
          showHeader: false,
          showFooter: false,
          html: renderTocPage(tocEntries)
        };
        if (tocIndex >= 0) sections.splice(tocIndex, 0, tocSection);
        else sections.push(tocSection);
      }

      state.sections = sections;
      return sections;
    }

    function renderChapterSection(chapter, isFirstChapter) {
      const pieces = [];
      pieces.push(`<div class="typeset">`);
      pieces.push(`<div class="chapter-start">`);
      pieces.push(`<div class="chapter-kicker">Chapter ${state.sections.filter(s => s.kind === "chapter").length + 1}</div>`);
      pieces.push(`<div class="chapter-title">${escHtml(chapter.title)}</div>`);
      pieces.push(`</div>`);

      let paragraphCount = 0;

      chapter.blocks.forEach(block => {
        if (block.type === "scene") {
          pieces.push(`<div class="scene-break">***</div>`);
          paragraphCount = 0;
          return;
        }

        if (block.type === "paragraph") {
          const noIndent = paragraphCount === 0;
          const dropcap = noIndent && isFirstChapter && els.enableDropCaps.checked ? " dropcap" : "";
          pieces.push(`<p class="${noIndent ? "no-indent" : ""}${dropcap}">${normalizeParagraphHtml(block.html)}</p>`);
          paragraphCount++;
        }
      });

      pieces.push(`</div>`);
      return pieces.join("");
    }

    function normalizeParagraphHtml(html) {
      const wrapper = document.createElement("div");
      wrapper.innerHTML = html;
      const text = wrapper.textContent?.trim() || "";

      if (!text && wrapper.querySelector("*")) {
        return wrapper.innerHTML;
      }

      if (wrapper.children.length === 1 && wrapper.firstElementChild.tagName.toLowerCase() === "p") {
        return wrapper.firstElementChild.innerHTML.trim();
      }

      if (!wrapper.children.length) {
        return escHtml(text).replace(/\n/g, "<br>");
      }

      return wrapper.innerHTML.trim();
    }

    function createMeasurePage(pageNumber, even = false) {

    function wordAwareAppend(container, html, pageEl, remainingBlocks) {
      const temp = document.createElement("div");
      temp.innerHTML = html;
      const node = temp.firstElementChild || temp.firstChild;

      if (!node) return false;

      container.appendChild(node);

      if (container.scrollHeight <= container.clientHeight) {
        return true;
      }

      container.removeChild(node);

      if (!(node instanceof HTMLElement) || node.tagName.toLowerCase() !== "p") {
        return false;
      }

      const fullText = node.innerHTML.trim();
      const words = fullText.split(/\s+/);
      if (words.length < 2) return false;

      const p = document.createElement("p");
      p.className = node.className;
      container.appendChild(p);

      let fitCount = 0;
      for (let i = 0; i < words.length; i++) {
        p.innerHTML += (i ? " " : "") + words[i];
        if (container.scrollHeight > container.clientHeight) {
          p.innerHTML = p.innerHTML.replace(new RegExp(`\\s?${escapeRegExp(words[i])}$`), "");
          break;
        }
        fitCount = i + 1;
      }

      if (fitCount === 0) {
        container.removeChild(p);
        return false;
      }

      const leftover = words.slice(fitCount).join(" ");
      if (leftover.trim()) {
        remainingBlocks.unshift(`<p class="${node.className}">${leftover}</p>`);
      }

      return true;
    }

    function escapeRegExp(string) {
      return string.replace(/[.*+?^${}()|[\]\\]/g, "\\$&");
    }

    function applyHyphenationFallback(root) {
      if (!els.enableHyphenation.checked) return;

      const walker = document.createTreeWalker(root, NodeFilter.SHOW_TEXT);
      const textNodes = [];
      while (walker.nextNode()) textNodes.push(walker.currentNode);

      textNodes.forEach(node => {
        const parentTag = node.parentElement?.tagName?.toLowerCase();
        if (["script", "style"].includes(parentTag)) return;

        node.textContent = node.textContent.replace(/\b[A-Za-z]{10,}\b/g, word => insertSoftHyphens(word));
      });
    }

    function insertSoftHyphens(word) {
      if (word.includes("\u00AD")) return word;

      // Lightweight fallback, still much better than no break opportunities.
      const vowels = "aeiouyAEIOUY";
      let out = "";
      for (let i = 0; i < word.length; i++) {
        out += word[i];
        const prev = word[i - 1] || "";
        const cur = word[i] || "";
        const next = word[i + 1] || "";
        const next2 = word[i + 2] || "";

        const goodBoundary =
          i > 2 &&
          i < word.length - 3 &&
          vowels.includes(cur) &&
          !vowels.includes(next) &&
          vowels.includes(next2 || "");

        if (goodBoundary) out += "\u00AD";
      }
      return out;
    }

    function applyJustificationTuning(pageContentBox) {
      if (!els.enableJustificationTuning.checked) return;

      const paragraphs = pageContentBox.querySelectorAll("p");
      paragraphs.forEach(p => {
        const text = p.textContent.trim();
        const words = text.split(/\s+/).filter(Boolean);
        if (!words.length) return;

        const avgLen = words.reduce((sum, w) => sum + w.length, 0) / words.length;

        let wordSpacing = 0;
        let letterSpacing = 0;

        if (avgLen < 4.3) wordSpacing = 0.02;
        if (avgLen > 6.5) wordSpacing = -0.01;
        if (words.length < 7) letterSpacing = -0.004;

        p.style.wordSpacing = `${wordSpacing}em`;
        p.style.letterSpacing = `${letterSpacing}em`;
      });
    }

    function paginateSections() {
      els.previewStack.innerHTML = "";
      els.hiddenMeasure.innerHTML = "";

      applyConfigToRoot();
      const sections = buildSections();

      const previewPages = [];
      let globalPageNumber = 0;
      const tocMap = [];

      sections.forEach((section, sectionIndex) => {
        const remainingBlocks = splitSectionIntoBlocks(section.html);

        // Each front-matter section starts fresh; each chapter starts fresh.
        while (remainingBlocks.length) {
          globalPageNumber++;
          const even = globalPageNumber % 2 === 0;

          const measurePage = createMeasurePage(globalPageNumber, even);
          const contentBox = measurePage.querySelector(".content-box");
          const header = measurePage.querySelector(".header");
          const footer = measurePage.querySelector(".footer");

          if (!section.showHeader || globalPageNumber === 1 || section.kind === "front") {
            header.style.visibility = "hidden";
          } else {
            header.querySelector(".header-left").textContent = even ? section.headerLeft : "";
            header.querySelector(".header-right").textContent = !even ? section.headerRight : "";
          }

          if (!section.showFooter || section.kind === "front") {
            footer.textContent = "";
          } else {
            footer.textContent = globalPageNumber;
          }

          els.hiddenMeasure.appendChild(measurePage);

          if (section.kind === "chapter" && remainingBlocks.length) {
            // Record first rendered page for TOC.
            if (!tocMap.some(t => t.id === section.id)) {
              tocMap.push({ id: section.id, title: section.chapterTitle, page: globalPageNumber });
            }
          }

          while (remainingBlocks.length) {
            const nextHtml = remainingBlocks.shift();
            const fit = wordAwareAppend(contentBox, nextHtml, measurePage, remainingBlocks);
            if (!fit) break;
          }

          if (els.enableHyphenation.checked) applyHyphenationFallback(contentBox);
          applyJustificationTuning(contentBox);

          const preview = createPreviewPage(globalPageNumber, even);
          syncPageContent(measurePage, preview.page);

          previewPages.push(preview.outer);
          els.hiddenMeasure.removeChild(measurePage);

          // If this was a front page like title/copyright/toc and it fit in one page, stop for that section.
          if (!remainingBlocks.length) break;
        }
      });

      // Fill TOC placeholders
      fillTocNumbers(previewPages, tocMap);

      previewPages.forEach(page => els.previewStack.appendChild(page));
      els.pageCount.textContent = String(previewPages.length);
      state.pages = previewPages;

      setStatus(`Preview ready • ${previewPages.length} pages`, "success");
    }

    function splitSectionIntoBlocks(sectionHtml) {
      const wrapper = document.createElement("div");
      wrapper.innerHTML = sectionHtml;
      const typeset = wrapper.firstElementChild || wrapper;

      // front-page should stay whole
      if (typeset.classList?.contains("front-page") || typeset.classList?.contains("copyright-page") || typeset.classList?.contains("toc-page")) {
        return [wrapper.innerHTML];
      }

      const blocks = [];
      Array.from(typeset.childNodes).forEach(node => {
        if (node.nodeType === Node.TEXT_NODE && !node.textContent.trim()) return;
        if (node.nodeType === Node.ELEMENT_NODE) {
          blocks.push(node.outerHTML);
        }
      });
      return blocks;
    }

    function syncPageContent(sourcePage, targetPage) {
      targetPage.className = sourcePage.className;
      targetPage.innerHTML = sourcePage.innerHTML;
    }

    function fillTocNumbers(previewPages, tocMap) {
      if (!tocMap.length) return;

      previewPages.forEach(pageOuter => {
        const page = pageOuter.querySelector(".page");
        const contentBox = page.querySelector(".content-box");
        if (!contentBox) return;

        const tocSpans = contentBox.querySelectorAll("[data-toc-page]");
        if (!tocSpans.length) return;

        tocSpans.forEach((span, idx) => {
          if (tocMap[idx]) span.textContent = tocMap[idx].page;
        });
      });
    }

    async function generatePreview() {
      try {
        setStatus("Generating preview…", "warning");

        if (els.bodyFontUpload.files[0]) {
          await embedFont(els.bodyFontUpload.files[0], "body", "AlysumBodyEmbedded");
        }
        if (els.displayFontUpload.files[0]) {
          await embedFont(els.displayFontUpload.files[0], "display", "AlysumDisplayEmbedded");
        }

        paginateSections();
      } catch (error) {
        console.error(error);
        setStatus("Preview failed", "error");
      }
    }

    function exportPdfPrint() {
      if (!state.pages.length) {
        setStatus("Generate preview first", "warning");
        return;
      }
      const win = window.open("", "_blank");
      win.document.open();
      win.document.write(buildPrintableHtml());
      win.document.close();
      win.focus();
      setTimeout(() => win.print(), 300);
    }

    function safeXml(text) {
      return String(text ?? "")
        .replaceAll("&", "&amp;")
        .replaceAll("<", "&lt;")
        .replaceAll(">", "&gt;")
        .replaceAll('"', "&quot;")
        .replaceAll("'", "&apos;");
    }

    function sectionToXhtml(section) {
      return `<?xml version="1.0" encoding="utf-8"?>
<!DOCTYPE html>
<html xmlns="http://www.w3.org/1999/xhtml" lang="en" xml:lang="en">
<head>
  <meta charset="utf-8"/>
  <title>${safeXml(section.chapterTitle || els.bookTitle.value)}</title>
  <link rel="stylesheet" type="text/css" href="styles/book.css"/>
</head>
<body>
${section.html}
</body>
</html>`;
    }

    async function exportEpub() {
      if (!state.sections.length) {
        setStatus("Generate preview first", "warning");
        return;
      }
      if (!window.JSZip) {
        setStatus("JSZip not loaded", "error");
        return;
      }

      try {
        setStatus("Building EPUB…", "warning");

        const zip = new JSZip();
        const metaInf = zip.folder("META-INF");
        const oebps = zip.folder("OEBPS");
        const textFolder = oebps.folder("text");
        const stylesFolder = oebps.folder("styles");
        const fontsFolder = oebps.folder("fonts");

        zip.file("mimetype", "application/epub+zip", { compression: "STORE" });
        metaInf.file("container.xml", `<?xml version="1.0" encoding="UTF-8"?>
<container version="1.0" xmlns="urn:oasis:names:tc:opendocument:xmlns:container">
  <rootfiles>
    <rootfile full-path="OEBPS/package.opf" media-type="application/oebps-package+xml"/>
  </rootfiles>
</container>`);

        let fontCss = "";
        if (els.bodyFontUpload.files[0]) {
          const bodyFile = els.bodyFontUpload.files[0];
          const bodyBuffer = await bodyFile.arrayBuffer();
          fontsFolder.file(bodyFile.name, bodyBuffer);
          fontCss += `
@font-face{
  font-family:"EPUBBody";
  src:url("../fonts/${bodyFile.name}");
}
`;
        }

        if (els.displayFontUpload.files[0]) {
          const displayFile = els.displayFontUpload.files[0];
          const displayBuffer = await displayFile.arrayBuffer();
          fontsFolder.file(displayFile.name, displayBuffer);
          fontCss += `
@font-face{
  font-family:"EPUBDisplay";
  src:url("../fonts/${displayFile.name}");
}
`;
        }

        stylesFolder.file("book.css", `
${fontCss}
body{
  margin:5%;
  font-family:${els.bodyFontUpload.files[0] ? '"EPUBBody",' : ''} ${presetToStack(els.bodyFontPreset.value)};
  font-size:${parseFloat(els.fontSize.value || 11)}pt;
  line-height:${parseFloat(els.lineHeight.value || 1.48)};
  color:#111;
}
p{
  margin:0 0 .9em 0;
  text-indent:${parseFloat(els.indentSize.value || 1.2)}em;
  -webkit-hyphens:${els.enableHyphenation.checked ? 'auto' : 'manual'};
  hyphens:${els.enableHyphenation.checked ? 'auto' : 'manual'};
}
p.no-indent{ text-indent:0; }
.chapter-kicker{
  text-align:center;
  text-transform:uppercase;
  letter-spacing:.16em;
  font-size:10pt;
  margin:0 0 1.2em;
  color:#555;
}
.chapter-title{
  text-align:center;
  font-family:${els.displayFontUpload.files[0] ? '"EPUBDisplay",' : ''} ${presetToStack(els.displayFontPreset.value)};
  font-size:20pt;
  margin:0 0 1.6em;
  line-height:1.2;
}
.scene-break{
  text-align:center;
  letter-spacing:.2em;
  margin:1.2em 0;
}
.front-page{
  text-align:center;
  margin-top:35%;
}
.title-page-title{
  font-family:${els.displayFontUpload.files[0] ? '"EPUBDisplay",' : ''} ${presetToStack(els.displayFontPreset.value)};
  font-size:24pt;
  line-height:1.18;
}
.title-page-subtitle{ margin-top:12pt; color:#555; }
.title-page-author{ margin-top:25%; text-transform:uppercase; letter-spacing:.08em; }
.toc-title{
  text-align:center;
  font-size:18pt;
  margin-bottom:1em;
}
.toc-row{
  display:flex;
  justify-content:space-between;
  border-bottom:1px dotted #999;
  margin-bottom:.5em;
}
        `.trim());

        const manifestItems = [];
        const spineItems = [];
        const navPoints = [];

        state.sections.forEach((section, index) => {
          const filename = `section-${String(index + 1).padStart(3, "0")}.xhtml`;
          textFolder.file(filename, sectionToXhtml(section));
          manifestItems.push(`<item id="item-${index + 1}" href="text/${filename}" media-type="application/xhtml+xml"/>`);
          spineItems.push(`<itemref idref="item-${index + 1}"/>`);
          navPoints.push(`<li><a href="text/${filename}">${safeXml(section.chapterTitle || section.id)}</a></li>`);
        });

        textFolder.file("nav.xhtml", `<?xml version="1.0" encoding="utf-8"?>
<!DOCTYPE html>
<html xmlns="http://www.w3.org/1999/xhtml" xmlns:epub="http://www.idpf.org/2007/ops" lang="en">
<head>
  <meta charset="utf-8"/>
  <title>Table of Contents</title>
</head>
<body>
  <nav epub:type="toc" id="toc">
    <h1>Contents</h1>
    <ol>
      ${navPoints.join("\n")}
    </ol>
  </nav>
</body>
</html>`);

        manifestItems.push(`<item id="nav" href="text/nav.xhtml" media-type="application/xhtml+xml" properties="nav"/>`);
        manifestItems.push(`<item id="css" href="styles/book.css" media-type="text/css"/>`);

        if (els.bodyFontUpload.files[0]) {
          manifestItems.push(`<item id="font-body" href="fonts/${els.bodyFontUpload.files[0].name}" media-type="${guessFontMediaType(els.bodyFontUpload.files[0].name)}"/>`);
        }
        if (els.displayFontUpload.files[0]) {
          manifestItems.push(`<item id="font-display" href="fonts/${els.displayFontUpload.files[0].name}" media-type="${guessFontMediaType(els.displayFontUpload.files[0].name)}"/>`);
        }

        oebps.file("package.opf", `<?xml version="1.0" encoding="utf-8"?>
<package version="3.0" unique-identifier="bookid" xmlns="http://www.idpf.org/2007/opf">
  <metadata xmlns:dc="http://purl.org/dc/elements/1.1/">
    <dc:identifier id="bookid">${safeXml(els.bookIsbn.value || `alysum-${Date.now()}`)}</dc:identifier>
    <dc:title>${safeXml(els.bookTitle.value || "Untitled")}</dc:title>
    <dc:creator>${safeXml(els.bookAuthor.value || "")}</dc:creator>
    <dc:language>en</dc:language>
    <dc:publisher>${safeXml(els.bookPublisher.value || "Alysum")}</dc:publisher>
    <dc:date>${safeXml(String(els.bookYear.value || new Date().getFullYear()))}</dc:date>
  </metadata>
  <manifest>
    ${manifestItems.join("\n")}
  </manifest>
  <spine>
    ${spineItems.join("\n")}
  </spine>
</package>`);

        const blob = await zip.generateAsync({ type: "blob" });
        downloadBlob(blob, `${slugify(els.bookTitle.value || "book")}.epub`);
        setStatus("EPUB exported", "success");
      } catch (error) {
        console.error(error);
        setStatus("EPUB export failed", "error");
      }
    }

    function guessFontMediaType(filename) {
      const lower = filename.toLowerCase();
      if (lower.endsWith(".woff2")) return "font/woff2";
      if (lower.endsWith(".woff")) return "font/woff";
      if (lower.endsWith(".otf")) return "font/otf";
      return "font/ttf";
    }

    function downloadBlob(blob, filename) {
      const url = URL.createObjectURL(blob);
      const a = document.createElement("a");
      a.href = url;
      a.download = filename;
      document.body.appendChild(a);
      a.click();
      a.remove();
      setTimeout(() => URL.revokeObjectURL(url), 500);
    }

    function saveProjectJson() {
      const blob = new Blob([JSON.stringify(getProjectData(), null, 2)], { type: "application/json" });
      downloadBlob(blob, `${slugify(els.bookTitle.value || "book-project")}.json`);
    }

    async function handleLoadProject(file) {
      const text = await file.text();
      const project = JSON.parse(text);
      loadProjectData(project);
      setStatus("Project loaded", "success");
    }
