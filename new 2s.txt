    function escapeHtml(text = "") {
      return text
        .replace(/&/g, "&amp;")
        .replace(/</g, "&lt;")
        .replace(/>/g, "&gt;");
    }

    function normalizeTitle(str = "") {
      return str.toLowerCase().replace(/[^a-z0-9]/g, "").trim();
    }

    function extractWikiLinks(text = "") {
      const matches = [...text.matchAll(/\[\[(.*?)\]\]/g)];
      return [...new Set(matches.map(m => (m[1] || "").trim()).filter(Boolean))];
    }

    function extractTags(text = "") {
      const matches = text.match(/(^|\s)#([a-zA-Z0-9_-]+)/g) || [];
      return [...new Set(matches.map(tag => tag.trim().replace(/^.*#/, "#")))];
    }

    function stripWiki(text = "") {
      return text.replace(/\[\[(.*?)\]\]/g, "$1");
    }

    function previewText(text = "", length = 90) {
      const clean = stripWiki(text).replace(/\s+/g, " ").trim();
      return clean.length > length ? clean.slice(0, length) + "…" : clean || "Empty note";
    }

    function wordCount(text = "") {
      const clean = stripWiki(text).replace(/\s+/g, " ").trim();
      if (!clean) return 0;
      return clean.split(/\s+/).length;
    }

    function getAutoLinkTitleMap() {
      const map = new Map();
      notes.forEach(note => {
        const title = (note.title || "").trim();
        if (title) {
          map.set(normalizeTitle(title), title);
        }
      });
      return map;
    }

    function buildRenderedEditorHTML(text = "") {
      if (!text) return "";

      const noteMap = getAutoLinkTitleMap();
      const tokenRegex = /(\[\[[^\]]+\]\]|#[a-zA-Z0-9_-]+|\b[a-zA-Z0-9_-]+\b)/g;

      return escapeHtml(text).replace(tokenRegex, (token) => {
        if (token.startsWith("[[") && token.endsWith("]]")) {
          const label = token.slice(2, -2).trim();
          return `<span class="editor-link" data-note="${escapeHtml(label)}">${escapeHtml(token)}</span>`;
        }

        if (token.startsWith("#")) {
          return `<span class="editor-tag">${escapeHtml(token)}</span>`;
        }

        const match = noteMap.get(normalizeTitle(token));
        if (match) {
          return `<span class="editor-link" data-note="${escapeHtml(match)}">${escapeHtml(token)}</span>`;
        }

        if (token.startsWith("/")) {
          return `<span class="editor-command">${escapeHtml(token)}</span>`;
        }

        return escapeHtml(token);
      }).replace(/\n/g, "<br>");
    }

    function sortNotes(list) {
      const next = [...list];
      if (sortMode === "alpha") {
        next.sort((a, b) => (a.title || "").localeCompare(b.title || ""));
      } else {
        next.sort((a, b) => {
          if (!!b.pinned !== !!a.pinned) return b.pinned ? 1 : -1;
          return (b.updatedAt || 0) - (a.updatedAt || 0);
        });
      }
      return next;
    }

    function sortFolders(list) {
      return [...list].sort((a, b) => (a.name || "").localeCompare(b.name || ""));
    }

    function getFilteredNotes() {
      const q = filterText.trim().toLowerCase();
      if (!q) return notes;

      return notes.filter(note => {
        const hay = `${note.title || ""} ${note.body || ""}`.toLowerCase();
        return hay.includes(q);
      });
    }

    function getSearchHits() {
      const q = filterText.trim().toLowerCase();
      if (!q) return [];
      return sortNotes(getFilteredNotes()).slice(0, 12);

    function buildFolderNode(folder) {
      const wrap = document.createElement("div");
      wrap.className = "tree-group";

      const open = folderOpenState[folder.id] !== false;
      const childFolders = sortFolders(folderChildren(folder.id));
      const childNotes = sortNotes(notesInFolder(folder.id).filter(n => !n.pinned));

      const header = document.createElement("div");
      header.className = "tree-folder" + (activeFolderId === folder.id ? " active" : "");
      header.draggable = true;

      header.innerHTML = `
        <div class="folder-caret">${open ? "▼" : "▶"}</div>
        <div class="folder-name">📁 ${escapeHtml(folder.name || "Untitled Folder")}</div>
        <div class="folder-badge">${childFolders.length + childNotes.length}</div>
      `;

      header.addEventListener("click", () => {
        activeFolderId = folder.id;
        folderOpenState[folder.id] = !open;
        renderFolderTree();
      });

      header.addEventListener("dblclick", async () => {
        const name = prompt("Rename folder?", folder.name || "");
        if (!name) return;
        try {
          await updateDoc(folderDocRef(folder.id), {
            name: name.trim() || "Untitled Folder"
          });
          setStatusBrief("Folder renamed");
        } catch (err) {
          console.error(err);
          setStatus("Rename failed");
        }
      });

      header.addEventListener("dragstart", () => {
        draggedFolderId = folder.id;
      });

      header.addEventListener("dragend", () => {
        draggedFolderId = null;
      });

      header.addEventListener("dragover", e => {
        e.preventDefault();
        header.classList.add("drop-target");
      });

      header.addEventListener("dragleave", () => {
        header.classList.remove("drop-target");
      });

      header.addEventListener("drop", async e => {
        e.preventDefault();
        header.classList.remove("drop-target");

        try {
          if (draggedNoteId) {
            await updateDoc(noteDocRef(draggedNoteId), {
              folderId: folder.id,
              updatedAt: Date.now()
            });
          }

          if (draggedFolderId && draggedFolderId !== folder.id) {
            const bad = isDescendantFolder(folder.id, draggedFolderId);
            if (bad) {
              setStatusBrief("Invalid nested move", "Saved");
              return;
            }
            await updateDoc(folderDocRef(draggedFolderId), {
              parentId: folder.id
            });
          }

          renderSidebar();
        } catch (err) {
          console.error(err);
          setStatus("Move failed");
        }
      });

      wrap.appendChild(header);

      if (open) {
        const body = document.createElement("div");
        body.className = "tree-indent";
        childNotes.forEach(note => body.appendChild(buildNoteItem(note)));
        childFolders.forEach(child => body.appendChild(buildFolderNode(child)));
        wrap.appendChild(body);
      }

      return wrap;
    }

    function isDescendantFolder(targetId, candidateAncestorId) {
      let current = findFolderById(targetId);
      while (current) {
        if ((current.parentId || null) === candidateAncestorId) return true;
        current = current.parentId ? findFolderById(current.parentId) : null;
      }
      return false;
    }

    function renderInlineChips() {
      const note = activeNote();

      inlineLinks.innerHTML = "";
      inlineTags.innerHTML = "";

      if (!note) {

        return;

      const tags = extractTags(note.body || "");

      if (!links.length) {
        inlineLinks.innerHTML = `<div class="muted">No links in this note.</div>`;
      } else {
        links.forEach(link => {
          const chip = document.createElement("button");
          chip.className = "link-chip";
          chip.type = "button";
          chip.textContent = `[[${link}]]`;
          chip.addEventListener("click", async () => {

          });
          inlineLinks.appendChild(chip);
        });
      }

      if (!tags.length) {
        inlineTags.innerHTML = `<div class="muted">No tags yet.</div>`;
      } else {
        tags.forEach(tag => {
          const chip = document.createElement("button");
          chip.className = "tag-chip";
          chip.type = "button";
          chip.textContent = tag;
          chip.addEventListener("click", () => {
            searchInput.value = tag;
            filterText = tag.toLowerCase();

            renderSearchResults();
          });
          inlineTags.appendChild(chip);
        });
      }
    }

    async function createNote(title = "Untitled", folderId = null) {
      const payload = {
        title,
        body: "",
        folderId,
        pinned: false,
        createdAt: Date.now(),
        updatedAt: Date.now()
      };

      const ref = await addDoc(notesRef(), payload);
      return { id: ref.id, ...payload };
    }

    async function createFolder(name = "New Folder", parentId = null) {
      const payload = {
        name,
        parentId,
        createdAt: Date.now()
      };

      const ref = await addDoc(foldersRef(), payload);
      folderOpenState[ref.id] = true;
      return { id: ref.id, ...payload };
    }

    async function updateBacklinksForRename(oldTitle, newTitle, skipNoteId) {
      const oldNorm = normalizeTitle(oldTitle);
      const newLabel = newTitle.trim();

      if (!oldNorm || oldNorm === normalizeTitle(newLabel)) return;

      const batch = writeBatch(db);
      let changed = 0;

      notes.forEach(note => {
        if (note.id === skipNoteId) return;
        const body = note.body || "";
        const updated = body.replace(/\[\[(.*?)\]\]/g, (match, inner) => {
          return normalizeTitle(inner) === oldNorm ? `[[${newLabel}]]` : match;
        });

        if (updated !== body) {
          batch.update(noteDocRef(note.id), {
            body: updated,
            updatedAt: Date.now()
          });
          changed += 1;
        }
      });

      if (changed > 0) {
        await batch.commit();
      }
    }

    async function updateActiveNote(fields) {
      if (!activeNoteId) return;
      const note = activeNote();
      const oldTitle = note?.title || "";

      await updateDoc(noteDocRef(activeNoteId), {
        ...fields,
        updatedAt: Date.now()
      });

      if (typeof fields.title === "string" && normalizeTitle(fields.title) !== normalizeTitle(oldTitle)) {
        await updateBacklinksForRename(oldTitle, fields.title, activeNoteId);
      }
    }

    async function duplicateActiveNote() {
      const note = activeNote();
      if (!note) return;

      const title = `${note.title || "Untitled"} Copy`;
      const ref = await addDoc(notesRef(), {
        title,
        body: note.body || "",
        folderId: note.folderId || null,
        pinned: false,
        createdAt: Date.now(),
        updatedAt: Date.now()
      });

      activeNoteId = ref.id;
      setStatusBrief("Duplicated");
    }

    async function deleteActiveNote() {
      const note = activeNote();
      if (!note) return;

      if (!confirm(`Delete "${note.title || "Untitled"}"?`)) return;

      try {
        await deleteDoc(noteDocRef(note.id));
        activeNoteId = null;
        noteTitle.value = "";
        editorText.value = "";
        renderEditorDisplay();
        setStatusBrief("Deleted", "Ready");
      } catch (err) {
        console.error(err);
        setStatus("Delete failed");
      }
    }

    function autoLinkMentions(rawText = "") {
      if (!rawText) return rawText;

      const noteMap = getAutoLinkTitleMap();
      const explicitRanges = [];

      for (const match of rawText.matchAll(/\[\[(.*?)\]\]/g)) {
        explicitRanges.push([match.index, match.index + match[0].length]);
      }

      function isInsideExplicitLink(index) {
        return explicitRanges.some(([start, end]) => index >= start && index < end);
      }

      return rawText.replace(/\b([A-Za-z0-9_-]+)\b/g, (word, capture, offset) => {
        if (isInsideExplicitLink(offset)) return word;
        const existing = noteMap.get(normalizeTitle(word));
        if (!existing) return word;
        return `[[${existing}]]`;
      });
    }

    function scheduleSave() {
      if (!activeNoteId) return;

      clearTimeout(saveTimer);
      setStatus("Saving...");

      saveTimer = setTimeout(async () => {
        try {
          const linkedBody = autoLinkMentions(editorText.value);

          if (linkedBody !== editorText.value) {
            const caret = editorText.selectionStart;
            editorText.value = linkedBody;
            editorText.setSelectionRange(caret, caret);
            renderEditorDisplay();
          }

          await updateActiveNote({
            title: noteTitle.value.trim() || "Untitled",
            body: editorText.value
          });
          setStatus("Saved");
        } catch (err) {
          console.error(err);
          setStatus("Save failed");
        }
      }, 500);
    }

    function insertTextAtCursor(text, replace = null) {
      const start = editorText.selectionStart;
      const end = editorText.selectionEnd;
      const value = editorText.value;

      if (replace) {
        const before = value.slice(0, replace.start);
        const after = value.slice(replace.end);
        editorText.value = before + text + after;
        const pos = before.length + text.length;
        editorText.setSelectionRange(pos, pos);
      } else {
        editorText.value = value.slice(0, start) + text + value.slice(end);
        const pos = start + text.length;
        editorText.setSelectionRange(pos, pos);
      }

      editorText.focus();
      renderEditorDisplay();
      scheduleSave();
      updateSuggestionState();
      renderRightPanel();
    }

    async function openOrCreateNote(title) {
      const clean = title.trim();
      let existing = notes.find(n => normalizeTitle(n.title) === normalizeTitle(clean));

      if (existing) {
        openNote(existing.id);
        return;
      }

      const ref = await addDoc(notesRef(), {
        title: clean || "Untitled",
        body: "",
        folderId: activeFolderId || null,
        pinned: false,
        createdAt: Date.now(),
        updatedAt: Date.now()
      });

      activeNoteId = ref.id;
      setStatusBrief("Created linked note");
      if (isMobile()) {
        setMobileView("editor");
      }
    }

    async function copyCurrentNote() {
      const note = activeNote();
      if (!note) return;

      const text = `# ${note.title || "Untitled"}\n\n${note.body || ""}`;
      try {
        await navigator.clipboard.writeText(text);
        setStatusBrief("Copied note");
      } catch {
        setStatusBrief("Copy failed", "Saved");
      }
    }

    async function copyCurrentLink() {

        await navigator.clipboard.writeText(`[[${note.title || "Untitled"}]]`);
        setStatusBrief("Copied link");
      } catch {
        setStatusBrief("Copy failed", "Saved");
      }
    }

    function closeSuggestion() {
      suggestionState.open = false;
      suggestionState.mode = null;
      suggestionState.items = [];
      suggestionState.index = 0;
      suggestionState.query = "";
      suggestionState.matchRange = null;
      suggestionPop.classList.remove("open");
      suggestionPop.innerHTML = "";
    }

    function openSuggestion(mode, items, queryText, matchRange) {
      if (!items.length) {
        closeSuggestion();
        return;
      }

      suggestionState.open = true;
      suggestionState.mode = mode;
      suggestionState.items = items;
      suggestionState.index = 0;
      suggestionState.query = queryText || "";
      suggestionState.matchRange = matchRange;

      renderSuggestion();
    }

    function renderSuggestion() {
      if (!suggestionState.open || !suggestionState.items.length) {
        closeSuggestion();
        return;
      }

      suggestionPop.innerHTML = "";
      suggestionState.items.forEach((item, index) => {
        const div = document.createElement("div");
        div.className = "suggestion-item" + (index === suggestionState.index ? " active" : "");
        div.innerHTML = `
          <div class="suggestion-title">${escapeHtml(item.label)}</div>
          <div class="suggestion-sub">${escapeHtml(item.sub || "")}</div>
        `;
        div.addEventListener("mousedown", (e) => {
          e.preventDefault();
          applySuggestion(index);
        });
        suggestionPop.appendChild(div);
      });

      suggestionPop.classList.add("open");
    }

    function getLinkSuggestionSource(queryText) {
      const q = queryText.toLowerCase();
      const items = sortNotes([...notes])
        .filter(n => !q || (n.title || "").toLowerCase().includes(q))
        .slice(0, 8)
        .map(n => ({
          type: "wiki",
          value: n.title || "Untitled",
          label: n.title || "Untitled",
          sub: previewText(n.body || "", 60)
        }));

      if (!q || !items.find(i => normalizeTitle(i.value) === normalizeTitle(q))) {
        items.unshift({
          type: "wiki",
          value: queryText.trim() || "Untitled",
          label: `Create "${queryText.trim() || "Untitled"}"`,
          sub: "New linked note"
        });
      }

      return items;
    }

    function getSlashCommands(queryText) {
      const commands = [
        { type: "slash", value: "heading", label: "/heading", sub: "Insert a level-2 heading", insert: "## Heading" },
        { type: "slash", value: "todo", label: "/todo", sub: "Insert a task checkbox", insert: "- [ ] Todo" },
        { type: "slash", value: "quote", label: "/quote", sub: "Insert a block quote", insert: "> Quote" },
        { type: "slash", value: "link", label: "/link", sub: "Insert a wiki link", insert: "[[Note]]" },
        { type: "slash", value: "tag", label: "/tag", sub: "Insert a tag", insert: "#tag" }
      ];

      const q = queryText.toLowerCase();
      return commands.filter(c => !q || c.value.includes(q) || c.label.includes(q));
    }

    function getTextBeforeCaret() {
      return editorText.value.slice(0, editorText.selectionStart);
    }

    function updateSuggestionState() {
      const before = getTextBeforeCaret();

      const wikiMatch = before.match(/\[\[([^\]]*)$/);
      if (wikiMatch) {
        const start = editorText.selectionStart - wikiMatch[0].length;
        openSuggestion("wiki", getLinkSuggestionSource(wikiMatch[1]), wikiMatch[1], {
          start,
          end: editorText.selectionStart
        });
        return;
      }

      const slashMatch = before.match(/\/([a-z]*)$/i);
      if (slashMatch) {
        const start = editorText.selectionStart - slashMatch[0].length;
        openSuggestion("slash", getSlashCommands(slashMatch[1]), slashMatch[1], {
          start,
          end: editorText.selectionStart
        });
        return;
      }

      closeSuggestion();
    }

    function applySuggestion(index) {
      const item = suggestionState.items[index];
      if (!item) return;

      if (suggestionState.mode === "wiki") {
        insertTextAtCursor(`[[${item.value}]]`, suggestionState.matchRange);
        closeSuggestion();
        return;
      }

      if (suggestionState.mode === "slash") {
        insertTextAtCursor(item.insert, suggestionState.matchRange);
        closeSuggestion();
      }
    }

    function handleEditorDisplayClick(e) {
      const link = e.target.closest(".editor-link");
      if (!link) return;

      const noteName = link.dataset.note || link.textContent.replace(/^\[\[/, "").replace(/\]\]$/, "");
      openOrCreateNote(noteName);
    }

          if (e.key === "ArrowDown") {
            e.preventDefault();
            suggestionState.index = (suggestionState.index + 1) % suggestionState.items.length;
            renderSuggestion();
            return;
          }

          if (e.key === "ArrowUp") {
            e.preventDefault();
            suggestionState.index = (suggestionState.index - 1 + suggestionState.items.length) % suggestionState.items.length;
            renderSuggestion();
            return;
          }

          if (e.key === "Enter" || e.key === "Tab") {
            e.preventDefault();
            applySuggestion(suggestionState.index);
            return;
          }

          if (e.key === "Escape") {
            e.preventDefault();
            closeSuggestion();
            return;

    function subscribeCollections() {
      if (unsubscribeNotes) unsubscribeNotes();
      if (unsubscribeFolders) unsubscribeFolders();

      const notesQ = query(notesRef(), orderBy("updatedAt", "desc"));
      unsubscribeNotes = onSnapshot(notesQ, (snap) => {
        notes = snap.docs.map(d => ({ id: d.id, ...d.data() }));

        if (!notes.length) {
          activeNoteId = null;
          treeRoot.innerHTML = `<div class="empty-state">No notes yet. Create one to begin.</div>`;
          pinnedList.innerHTML = `<div class="empty-state">No pinned notes yet.</div>`;
          noteTitle.value = "";
          editorText.value = "";
          renderEditorDisplay();
          renderRightPanel();
          return;
        }

        const existingActive = notes.find(n => n.id === activeNoteId);
        if (!notes.length) {
          activeNoteId = null;
        } else if (!activeNoteId || !existingActive) {
          activeNoteId = notes[0].id;
        }

        const userIsEditing = document.activeElement === editorText || document.activeElement === noteTitle;
        if (!userIsEditing) {
          renderEditor();
        } else {
          renderSidebar();
          renderRightPanel();
          renderEditorDisplay();
        }
      });

      unsubscribeFolders = onSnapshot(foldersRef(), (snap) => {
        folders = snap.docs.map(d => ({ id: d.id, ...d.data() }));
        renderSidebar();
      });
    }
